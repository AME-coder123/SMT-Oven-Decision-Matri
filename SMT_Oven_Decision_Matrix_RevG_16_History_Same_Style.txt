<script>
const inputs=[...document.querySelectorAll('input[type=number]')];
const el=id=>document.getElementById(id);
const pcbInputs=inputs.filter(x=>+x.id.slice(1)<=8);
const asicInputs=inputs.filter(x=>+x.id.slice(1)>=9 && +x.id.slice(1)<=14);
const vpmInputs=inputs.filter(x=>+x.id.slice(1)>=15 && +x.id.slice(1)<=20);

function showP(x,n){
  document.querySelectorAll('.page').forEach(p=>p.classList.remove('active'));
  el(x).classList.add('active');
  document.querySelectorAll('.step').forEach((b,i)=>b.classList.toggle('active',i===n-1));
  window.scrollTo({top:0,behavior:'smooth'});
}
function s(v,a,b,c,d){if(!v)return 0;if(v<=a)return 20;if(v<=b)return 40;if(v<=c)return 70;if(v<=d)return 90;return 100}
function applied(name){return document.querySelector('input[name="'+name+'"]:checked')?.value==='yes'}

function setOptional(sectionId, fieldList, on, badgeId){
  el(sectionId).classList.toggle('disabled',!on);
  fieldList.forEach(i=>{i.disabled=!on;if(!on)i.value=''});
  el(badgeId).textContent=on?'Applicable':'Not Applicable';
}

function updateApplicability(){
  const a=applied('asicApply'), v=applied('vpmApply');
  setOptional('asicSection',asicInputs,a,'asicBadge');
  setOptional('vpmSection',vpmInputs,v,'vpmBadge');
  calc();
}


function renderOven(heatZones, coolZones, classification, thermal){
  const g=el('ovenZones');
  g.innerHTML='';
  const total=heatZones+coolZones;
  const left=82, right=918, y=92, h=105;
  const gap=2;
  const zoneW=Math.max(7,(right-left-gap*(total-1))/total);

  for(let i=0;i<total;i++){
    const heating=i<heatZones;
    const x=left+i*(zoneW+gap);
    const fill=heating?'#d92d20':'#1769aa';
    const opacity=heating?.20:.16;
    const strokeOpacity=heating?.70:.55;

    const r=document.createElementNS('http://www.w3.org/2000/svg','rect');
    r.setAttribute('x',x); r.setAttribute('y',y);
    r.setAttribute('width',zoneW); r.setAttribute('height',h);
    r.setAttribute('rx','2'); r.setAttribute('fill',fill);
    r.setAttribute('fill-opacity',opacity);
    r.setAttribute('stroke','currentColor'); r.setAttribute('stroke-opacity',strokeOpacity);
    g.appendChild(r);

    const t=document.createElementNS('http://www.w3.org/2000/svg','text');
    t.setAttribute('x',x+zoneW/2); t.setAttribute('y',y+57);
    t.setAttribute('text-anchor','middle'); t.setAttribute('font-size',zoneW>20?'9':'7');
    t.setAttribute('fill',heating?'#b42318':'#175ea8');
    t.setAttribute('font-weight','700');
    t.textContent=heating?'H'+(i+1):'C'+(i-heatZones+1);
    g.appendChild(t);
  }

  const dividerX=left+heatZones*(zoneW+gap)-gap/2;
  const line=document.createElementNS('http://www.w3.org/2000/svg','line');
  line.setAttribute('x1',dividerX); line.setAttribute('x2',dividerX);
  line.setAttribute('y1',y-18); line.setAttribute('y2',y+h+15);
  line.setAttribute('stroke','currentColor'); line.setAttribute('stroke-width','2');
  line.setAttribute('stroke-dasharray','5 4'); line.setAttribute('stroke-opacity','.55');
  g.appendChild(line);

  const ht=document.createElementNS('http://www.w3.org/2000/svg','text');
  ht.setAttribute('x',(left+dividerX)/2); ht.setAttribute('y','82');
  ht.setAttribute('text-anchor','middle'); ht.setAttribute('font-size','10');
  ht.setAttribute('font-weight','700'); ht.setAttribute('fill','currentColor');
  ht.textContent='HEATING ZONES';
  g.appendChild(ht);

  const ct=document.createElementNS('http://www.w3.org/2000/svg','text');
  ct.setAttribute('x',(dividerX+right)/2); ct.setAttribute('y','82');
  ct.setAttribute('text-anchor','middle'); ct.setAttribute('font-size','10');
  ct.setAttribute('font-weight','700'); ct.setAttribute('fill','currentColor');
  ct.textContent='COOLING ZONES';
  g.appendChild(ct);

  el('ovenVisualStatus').textContent=classification+' · '+heatZones+' heating / '+coolZones+' cooling';
  el('heatChip').textContent='Heating: '+heatZones+' zones';
  el('coolChip').textContent='Cooling: '+coolZones+' zones';
  el('thermalChip').textContent='Thermal Capacity: '+thermal;
  el('classChip').textContent='Classification: '+classification;
}

function calc(){
  const a=applied('asicApply'), v=applied('vpmApply');
  const active=[...pcbInputs,...(a?asicInputs:[]),...(v?vpmInputs:[])];
  const filled=active.filter(x=>x.value!=='').length, total=active.length;
  el('progress').style.width=(filled/total*100)+'%';
  el('progressText').textContent=filled+' / '+total+' applicable inputs';

  const pcb=Math.round((s(+i1.value,10,15,24,30)+s(+i2.value,10,15,24,30)+s(+i3.value,2.5,4,5.3,8))/3);
  const asic=a?Math.round((s(+i9.value,65,80,110,150)+s(+i10.value,65,80,110,150)+s(+i11.value,4,6,7,8))/3):0;
  const vpm=v?Math.round((s(+i15.value,40,50,65,100)+s(+i16.value,40,50,65,100)+s(+i17.value,4,6,7,8))/3):0;

  // Original 40/40/20 weights are preserved when all blocks apply.
  // If a block is not applicable, its weight is redistributed proportionally.
  const weights={pcb:.4}; if(a)weights.asic=.4; if(v)weights.vpm=.2;
  const sum=Object.values(weights).reduce((x,y)=>x+y,0);
  const t=Math.round(pcb*weights.pcb/sum+(a?asic*weights.asic/sum:0)+(v?vpm*weights.vpm/sum:0));
  const c=t<=30?'Low':t<=50?'Medium':t<=70?'High':t<=90?'Advanced':'Emerging Technologies';
  const qualified=filled===total;

  el('pcb').textContent=pcb; el('asic').textContent=a?asic:'N/A'; el('vpm').textContent=v?vpm:'N/A'; el('tot').textContent=t;
  el('score').textContent=t; el('clsHero').textContent=filled?c:'Waiting for Inputs'; el('ovenClass').textContent=filled?c:'Waiting';
  el('scoreFill').style.width=t+'%';
  el('qm').textContent=filled+' / '+total; el('qs').textContent=qualified?'Qualified':'Not Qualified';
  el('qc').textContent=qualified?'Qualified':'Not Qualified'; el('rs').textContent=qualified?'Recommended':'Qualification Incomplete';
  el('rsBadge').textContent=qualified?'Qualified':'Qualification Incomplete';

  const map={Low:['10/10','3','Standard'],Medium:['13/13','3','Enhanced'],High:['15/15','4','Advanced'],Advanced:['17/17','5','Ultra High'],'Emerging Technologies':['20/20','5','Future Ultra']};
  if(filled){
    el('hz').textContent=map[c][0];
    el('cz').textContent=map[c][1];
    el('tc').textContent=map[c][2];
    el('ni').textContent='Yes';
    el('cl').textContent='29-70 mm';
    renderOven(parseInt(map[c][0]),parseInt(map[c][1]),c,map[c][2]);
  } else {
    ['hz','cz','tc','ni','cl'].forEach(id=>el(id).textContent='Not Defined');
    el('ovenVisualStatus').textContent='Waiting for classification';
    el('heatChip').textContent='Heating: —';
    el('coolChip').textContent='Cooling: —';
    el('thermalChip').textContent='Thermal Capacity: —';
    el('classChip').textContent='Classification: —';
    el('ovenZones').innerHTML='';
  }

  const blocks=['PCB',...(a?['ASIC/BGA']:[]),...(v?['VPM']:[])].join(' · ');
  el('appBlocks').textContent=blocks; el('dashBlocks').textContent=blocks;
  [['b1',pcb],['b2',a?asic:0],['b3',v?vpm:0],['b4',t]].forEach(([id,val])=>{el(id).style.width=val+'%';el(id+'txt').textContent=(id==='b2'&&!a)||(id==='b3'&&!v)?'N/A':val+'%'});
  el('dashClass').textContent=filled?c:'Waiting'; el('dashClass2').textContent=filled?c:'Waiting for Inputs';

function saveConsultation(){
  const records=getHistory();
  const now=new Date();
  const seq=(records.length?Math.max(...records.map(r=>Number((r.id||'').split('-').pop())||0)):0)+1;
  const id='SMT-'+now.getFullYear()+'-'+String(seq).padStart(5,'0');
  const record={
    id, timestamp:now.toISOString(), product:el('productName')?.value.trim()||'Unnamed Assembly',
    inputs:collectInputs(), result:currentResult()
  };
  records.unshift(record);
  setHistory(records);
  renderHistory();
  showP('history',6);
  const saveBtn=el('saveConsultationBtn');
  if(saveBtn){
    const oldText=saveBtn.innerHTML;
    saveBtn.innerHTML='✓ Saved: '+id;
    setTimeout(()=>saveBtn.innerHTML=oldText,2500);
  }
}
function renderHistory(){
  const q=(el('historySearch')?.value||'').toLowerCase().trim();
  const cls=el('historyClass')?.value||'';
  const product=(el('historyProduct')?.value||'').toLowerCase().trim();
  const all=getHistory();
  const rows=all.filter(r=>
    (!q || [r.id,r.product,r.result.classification,r.result.score].join(' ').toLowerCase().includes(q)) &&
    (!cls || r.result.classification===cls) &&
    (!product || r.product.toLowerCase().includes(product))
  );
  el('historyCount').textContent=rows.length+' record'+(rows.length===1?'':'s')+(all.length!==rows.length?' · '+all.length+' total':'');
  if(!rows.length){el('historyTable').innerHTML='<div class="history-empty">No saved consultations match the current filters.</div>';return}
  let out='<table class="history-table"><thead><tr><th>ID</th><th>Date</th><th>Product / Assembly</th><th>Score</th><th>Classification</th><th>Oven</th><th></th></tr></thead><tbody>';
  rows.forEach(r=>{
    const d=new Date(r.timestamp);
    out+=`<tr><td><strong>${esc(r.id)}</strong></td><td>${esc(d.toLocaleString())}</td><td>${esc(r.product)}</td><td><strong>${esc(r.result.score)}</strong></td><td>${esc(r.result.classification)}</td><td>${esc(r.result.heatingZones)} H / ${esc(r.result.coolingZones)} C</td><td><div class="history-actions"><button class="btn" onclick="viewHistory('${escAttr(r.id)}')">View</button><button class="btn" onclick="duplicateHistory('${escAttr(r.id)}')">Duplicate</button><button class="btn danger" onclick="deleteHistory('${escAttr(r.id)}')">Delete</button></div></td></tr>`;
  });
  out+='</tbody></table>'; el('historyTable').innerHTML=out;
}
